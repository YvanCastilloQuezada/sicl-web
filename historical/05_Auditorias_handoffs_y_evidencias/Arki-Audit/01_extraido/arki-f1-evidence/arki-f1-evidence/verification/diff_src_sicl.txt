[32m+[m[32m"""Explicit, auditable recomputation handler registry for H-005."""[m
[32m+[m[32mfrom __future__ import annotations[m
[32m+[m
[32m+[m[32mfrom dataclasses import dataclass[m
[32m+[m[32mfrom typing import Callable, Mapping[m
[32m+[m
[32m+[m[32mfrom ..derivation import DerivationRecord, VersionedRef[m
[32m+[m
[32m+[m
[32m+[m[32mclass RegistryError(Exception):[m
[32m+[m[32m    pass[m
[32m+[m
[32m+[m
[32m+[m[32mclass HandlerNotRegisteredError(RegistryError):[m
[32m+[m[32m    pass[m
[32m+[m
[32m+[m
[32m+[m[32mclass AmbiguousHandlerError(RegistryError):[m
[32m+[m[32m    pass[m
[32m+[m
[32m+[m
[32m+[m[32m@dataclass(frozen=True, order=True)[m
[32m+[m[32mclass HandlerKey:[m
[32m+[m[32m    method: str[m
[32m+[m[32m    method_version: str[m
[32m+[m
[32m+[m
[32m+[m[32mRecomputationHandler = Callable[[str, DerivationRecord, Mapping[tuple[str, str], VersionedRef]], DerivationRecord][m
[32m+[m
[32m+[m
[32m+[m[32mclass RecomputationRegistry:[m
[32m+[m[32m    """Only explicitly registered callables can be resolved."""[m
[32m+[m
[32m+[m[32m    def __init__(self) -> None:[m
[32m+[m[32m        self._handlers: dict[HandlerKey, tuple[RecomputationHandler, ...]] = {}[m
[32m+[m
[32m+[m[32m    def register(self, method: str, method_version: str, handler: RecomputationHandler) -> None:[m
[32m+[m[32m        if not isinstance(method, str) or not method.strip() or not isinstance(method_version, str) or not method_version.strip():[m
[32m+[m[32m            raise RegistryError("method and method_version must be non-empty strings")[m
[32m+[m[32m        if not callable(handler):[m
[32m+[m[32m            raise RegistryError("handler must be callable")[m
[32m+[m[32m        key = HandlerKey(method.strip(), method_version.strip())[m
[32m+[m[32m        existing = self._handlers.get(key, ())[m
[32m+[m[32m        self._handlers[key] = existing + (handler,)[m
[32m+[m
[32m+[m[32m    def resolve(self, method: str, method_version: str) -> RecomputationHandler:[m
[32m+[m[32m        key = HandlerKey(method, method_version)[m
[32m+[m[32m        handlers = self._handlers.get(key, ())[m
[32m+[m[32m        if not handlers:[m
[32m+[m[32m            raise HandlerNotRegisteredError(f"HANDLER_NOT_REGISTERED: {method}@{method_version}")[m
[32m+[m[32m        if len(handlers) != 1:[m
[32m+[m[32m            raise AmbiguousHandlerError(f"AMBIGUOUS_HANDLER: {method}@{method_version}")[m
[32m+[m[32m        return handlers[0][m
[32m+[m
[32m+[m[32m    def keys(self) -> tuple[HandlerKey, ...]:[m
[32m+[m[32m        return tuple(sorted(self._handlers))[m
[32m+[m
[32m+[m
[32m+[m[32m__all__ = ["RegistryError", "HandlerNotRegisteredError", "AmbiguousHandlerError", "HandlerKey", "RecomputationHandler", "RecomputationRegistry"][m
