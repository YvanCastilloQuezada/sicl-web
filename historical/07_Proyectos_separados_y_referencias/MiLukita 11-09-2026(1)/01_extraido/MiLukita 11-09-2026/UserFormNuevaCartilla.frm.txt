Private Sub CommandButtonAleatoria_Click()
    Dim numJugadas As Integer
    Dim i As Integer
    Dim numeros(1 To 36) As Boolean
    Dim Numero As Integer
    Dim comboBoxName As String, labelNameResumen As String
    
    ' Determinar cuántas jugadas se han seleccionado
    If CheckBox61.value Then numJugadas = 1
    If CheckBox62.value Then numJugadas = 2
    If CheckBox63.value Then numJugadas = 3
    If CheckBox64.value Then numJugadas = 4
    If CheckBox65.value Then numJugadas = 5
        
    ' Limpiar los ComboBoxes y Labels
    For i = 1 To 5
        Me.Controls("ComboBox" & i).value = ""
        Me.Controls("Label" & (i + 100)).Caption = ""
    Next i
    Me.ComboBox6.value = ""
    Me.Label106.Caption = ""
    
    ' Reiniciar los números generados
    For i = 1 To 36
        numeros(i) = False
    Next i

    ' Generar números aleatorios según el número de jugadas
    For i = 1 To numJugadas
        Do
            Numero = WorksheetFunction.RandBetween(1, 36)
        Loop While numeros(Numero)
        
        numeros(Numero) = True
        
        comboBoxName = "ComboBox" & i
        labelNameResumen = "Label" & (i + 100)
        
        On Error Resume Next
        Me.Controls(comboBoxName).value = Numero
        Me.Controls(labelNameResumen).Caption = Numero
        On Error GoTo 0
    Next i
    
    ' Generar un número aleatorio para ComboBox6 si se selecciona algún CheckBox del 61 al 65
    If numJugadas > 3 Then
        Dim Numero6 As Integer
        Dim numeros6(1 To 17) As Boolean
        
        ' Marcar números ya seleccionados en ComboBoxes 1 a 5
        For i = 1 To 5
            If Me.Controls("ComboBox" & i).value <> "" Then
                Numero = Me.Controls("ComboBox" & i).value
                If Numero <= 17 Then
                    numeros6(Numero) = True
                End If
            End If
        Next i
        
        ' Generar un número aleatorio entre 1 y 17 para ComboBox6 que no esté repetido
        Do
            Numero6 = WorksheetFunction.RandBetween(1, 17)
        Loop While numeros6(Numero6)
        
        Me.ComboBox6.value = Numero6
        Me.Label106.Caption = Numero6
    End If
End Sub

Private Sub ValidateComboBoxValue(changedComboBox As MSForms.ComboBox, label As MSForms.label)
    Dim valorSeleccionado As Integer
    Dim i As Integer
    Dim duplicado As Boolean
    duplicado = False
    
    If IsNumeric(changedComboBox.value) Then
        valorSeleccionado = CInt(changedComboBox.value)
        
        ' Verificar que el número no esté en ninguno de los otros ComboBoxes, excepto ComboBox6
        If changedComboBox.Name <> "ComboBox6" Then
            For i = 1 To 5
                If Me.Controls("ComboBox" & i).value = valorSeleccionado And Me.Controls("ComboBox" & i).Name <> changedComboBox.Name Then
                    duplicado = True
                    Exit For
                End If
            Next i
            
            If duplicado Then
                MsgBox "El número " & valorSeleccionado & " ya está seleccionado en otro ComboBox. Elija un número diferente.", vbExclamation
                changedComboBox.value = ""
            Else
                label.Caption = valorSeleccionado
            End If
        Else
            ' Si es ComboBox6, simplemente actualizar la etiqueta sin verificación de duplicado
            label.Caption = valorSeleccionado
        End If
    End If
End Sub

Private Sub InitializeComboBoxes()

    Dim cmb As MSForms.ComboBox

    ' Reiniciar números seleccionados

    Next i

    ' Inicializar los ComboBoxes con los valores apropiados
    For i = 1 To 5
        Set cmb = Me.Controls("ComboBox" & i)

    End If

Private Sub InitializeComboBox(cmb As MSForms.ComboBox, minVal As Integer, maxVal As Integer)

Sub CheckBox_Click(startIndex As Integer, endIndex As Integer, labelName As String)
    Dim i As Integer
    Dim selectedText As String
    selectedText = ""
    
    ' Recorrer todos los checkboxes en el rango
    For i = startIndex To endIndex
        If Me.Controls("CheckBox" & i).value = True Then
            ' Obtener el texto del checkbox seleccionado
            selectedText = Me.Controls("CheckBox" & i).Caption
            
            ' Desmarcar todos los otros checkboxes en el rango
            UncheckOtherCheckBoxes startIndex, endIndex, i
            
            ' Salir del bucle al encontrar el checkbox seleccionado
            Exit For
        End If
    Next i
    
    ' Actualizar el label correspondiente con el texto seleccionado
    If selectedText = "" And labelName = "labelMultiplicador" Then
        Me.Controls(labelName).Caption = "Sin Multiplicador"
    Else
        Me.Controls(labelName).Caption = selectedText
    End If
End Sub

Sub UncheckOtherCheckBoxes(startIndex As Integer, endIndex As Integer, selectedIdx As Integer)
    Dim i As Integer
    ' Recorrer todos los checkboxes en el rango y desmarcar los que no son el seleccionado
    For i = startIndex To endIndex
        If i <> selectedIdx Then
            Me.Controls("CheckBox" & i).value = False
        End If
    Next i
End Sub

Private Sub ActualizarMontoPagar()
    Dim multiplicador As Double
    Dim cantidadJugada As Integer
    Dim costoJugada As Double
    
    ' Inicializamos los valores
    multiplicador = 1
    cantidadJugada = 0
    costoJugada = CDbl(LabelCostoJugada.Caption)
    
    ' Revisar los CheckBoxes de multiplicador
    If CheckBox55.value = True Then
        multiplicador = 2
    ElseIf CheckBox56.value = True Then
        multiplicador = 3
    ElseIf CheckBox57.value = True Then
        multiplicador = 5
    ElseIf CheckBox58.value = True Then
        multiplicador = 10
    End If
    
    ' Revisar los CheckBoxes de cantidad de jugadas
    If CheckBox61.value = True Then
        cantidadJugada = 1
    ElseIf CheckBox62.value = True Then
        cantidadJugada = 2
    ElseIf CheckBox63.value = True Then
        cantidadJugada = 3
    ElseIf CheckBox64.value = True Then
        cantidadJugada = 4
    ElseIf CheckBox65.value = True Then
        cantidadJugada = 5
    End If
    
    ' Calcular el monto a pagar
    If cantidadJugada > 0 Then
        LabelMontoPagar.Caption = multiplicador * cantidadJugada * costoJugada
    Else
        LabelMontoPagar.Caption = 0
    End If
End Sub

Function GenerarNumeroUnico() As String
    Dim Numero As String
    Dim Letra1 As String, Letra2 As String, Letra3 As String, Letra4 As String
    Dim Digito As Integer
    Dim Intentos As Integer
    
    ' Inicializar la colección si es la primera vez
    If NumeroUsado Is Nothing Then
        Set NumeroUsado = New Collection
    End If
    
    Do
        Randomize
        Letra1 = Chr(Int((90 - 65 + 1) * Rnd + 65))  ' Genera una letra aleatoria entre A-Z
        Letra2 = Chr(Int((90 - 65 + 1) * Rnd + 65))  ' Genera una letra aleatoria entre A-Z
        Letra3 = Chr(Int((90 - 65 + 1) * Rnd + 65))  ' Genera una letra aleatoria entre A-Z
        Letra4 = Chr(Int((90 - 65 + 1) * Rnd + 65))  ' Genera una letra aleatoria entre A-Z
        Digito = Int((9 - 0 + 1) * Rnd + 0)          ' Genera un dígito aleatorio entre 0-9
        Numero = Letra1 & Letra2 & Letra3 & Letra4 & Digito
        
        ' Verificar que el número no se haya usado antes
        Intentos = Intentos + 1
        On Error Resume Next
        NumeroUsado.Add Numero, CStr(Numero)
        On Error GoTo 0
    Loop While Err.Number <> 0 And Intentos < 1000  ' Limita los intentos a 1000

    ' Devolver el número único generado
    GenerarNumeroUnico = Numero
End Function

Private Sub TextBoxBoleto_AfterUpdate()
    ' Validar y dar formato al boleto (4 letras y 1 número)
    Dim boleto As String
    boleto = TextBoxBoleto.value
    
    If Len(boleto) = 5 And IsNumeric(Right(boleto, 1)) And Not IsNumeric(Left(boleto, 4)) Then
        TextBoxBoleto.value = UCase(Left(boleto, 4)) & Right(boleto, 1)
    Else
        MsgBox "Por favor, ingrese 4 letras seguidas de 1 número.", vbExclamation
        TextBoxBoleto.SetFocus
    End If
    
    ' Llamar a la función para actualizar LabelBuscar
    ActualizarLabelBuscar
End Sub

Private Sub ActualizarLabelBuscar()
    ' Combinar los valores y mostrar en LabelBuscar
    If ComboBoxFecha.ListIndex <> -1 And ComboBoxHora.ListIndex <> -1 And _
       Len(TextBoxBoleto.value) = 5 And IsNumeric(Right(TextBoxBoleto.value, 1)) And _
       Not IsNumeric(Left(TextBoxBoleto.value, 4)) Then
       
        Dim FechaFormateada As String
        Dim FechaPartes() As String
        FechaPartes = Split(ComboBoxFecha.value, "-")
        FechaFormateada = FechaPartes(0) & "/" & FechaPartes(1) & "/" & FechaPartes(2) ' Convertir el formato a dd/mm/yyyy
        
        LabelBuscar.Caption = "Rul1-" & FechaFormateada & "-" & ComboBoxHora.value & "-" & TextBoxBoleto.value
    Else
        LabelBuscar.Caption = ""
    End If
End Sub

Private Sub ComboBoxFecha_Change()
    ' Llamar a la función ActualizarLabelBuscar al cambiar el valor de ComboBoxFecha
    ActualizarLabelBuscar
End Sub

Private Sub ComboBoxHora_Change()
    ' Llamar a la función ActualizarLabelBuscar al cambiar el valor de ComboBoxHora
    ActualizarLabelBuscar
End Sub

Private Sub TextBoxBoleto_Change()
    ' Llamar a la función ActualizarLabelBuscar al cambiar el valor de TextBoxBoleto
    ActualizarLabelBuscar
End Sub

' Función para verificar la existencia de un control en el formulario
Private Function ControlExists(controlName As String) As Boolean
    On Error Resume Next
    ControlExists = Not Me.Controls(controlName) Is Nothing
    On Error GoTo 0
End Function
