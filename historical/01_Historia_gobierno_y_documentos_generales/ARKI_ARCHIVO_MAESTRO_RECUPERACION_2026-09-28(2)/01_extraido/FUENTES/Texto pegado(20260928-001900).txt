"""A-002 detector: check_input_sufficiency. Pure, deterministic, fail-closed.

LIMIT: A-002 can reach at most EvidenceVerificationLevel.HASH_FORMAT_VALID.
CONTENT_HASH_MATCHES_DECLARED requires the actual content of the referenced
object, which is not passed to A-002. SUPPORT_SEMANTICALLY_VERIFIED is out
of scope by design.
"""
from __future__ import annotations

from typing import Any, Mapping

from .canonical import canonical_json, sha256_hex
from .contract import (
    ApplicabilityStatus,
    EpistemicState,
    EvaluationIdentityPayload,
    EvidenceRef,
    EvidenceVerificationLevel,
    FieldAssessment,
    FieldRequirement,
    HumanQuestion,
    NormativeApplicability,
    QuestionKind,
    SufficiencyProfile,
    SufficiencyReport,
    SufficiencyStatus,
)


def _has_content(value: Any) -> bool:
    """False and 0 are valid content. Empty strings/lists/dicts are not."""
    if value is None:
        return False
    if value is True or value is False:
        return True
    if isinstance(value, (int, float)) and value == 0:
        return True
    if isinstance(value, (str, list, tuple, dict, set)) and len(value) == 0:
        return False
    return True


def _fingerprint_snapshot(snapshot: Mapping[str, Any]) -> str:
    return sha256_hex(canonical_json(snapshot))


def _compute_evaluation_id(
    project_id: str,
    snapshot_fingerprint: str,
    profile: SufficiencyProfile,
) -> str:
    payload = EvaluationIdentityPayload(
        project_id=project_id,
        snapshot_fingerprint=snapshot_fingerprint,
        profile_id=profile.profile_id,
        profile_version=profile.profile_version,
        operation=profile.operation,
    )
    return sha256_hex(canonical_json(payload.to_dict()))


def _extract_declared_state(raw: Any) -> tuple[EpistemicState, tuple[str, ...]]:
    if raw is None:
        return EpistemicState.MISSING, ()
    if not isinstance(raw, Mapping):
        return EpistemicState.MISSING, ()
    ds = raw.get("declared_state")
    if not isinstance(ds, str):
        return EpistemicState.MISSING, ()
    try:
        state = EpistemicState(ds)
    except ValueError:
        return EpistemicState.MISSING, ()
    conflicts_raw = raw.get("conflicts")
    conflicts: tuple[str, ...] = ()
    if isinstance(conflicts_raw, (list, tuple)):
        conflicts = tuple(str(c) for c in conflicts_raw if c)
    if conflicts:
        return EpistemicState.CONFLICTING, conflicts
    return state, ()


def _extract_evidence(raw: Any) -> tuple[EvidenceRef, ...]:
    if not isinstance(raw, Mapping):
        return ()
    refs_raw = raw.get("evidence_refs")
    if not isinstance(refs_raw, (list, tuple)):
        return ()
    refs: list[EvidenceRef] = []
    for item in refs_raw:
        if not isinstance(item, Mapping):
            continue
        try:
            refs.append(EvidenceRef(
                id=str(item.get("id", "")),
                kind=str(item.get("kind", "")),
                content_hash=str(item.get("content_hash", "")),
                source_id=str(item.get("source_id", "")),
                declared_supports_field=str(item.get("declared_supports_field", "")),
            ))
        except (ValueError, TypeError):
            continue
    return tuple(refs)


def _extract_applicability(raw: Any) -> NormativeApplicability | None:
    if not isinstance(raw, Mapping):
        return None
    app_raw = raw.get("applicability")
    if not isinstance(app_raw, Mapping):
        return None
    try:
        return NormativeApplicability(
            jurisdiction_known=bool(app_raw.get("jurisdiction_known", False)),
            jurisdiction_id=app_raw.get("jurisdiction_id"),
            source_identified=bool(app_raw.get("source_identified", False)),
            source_id=app_raw.get("source_id"),
            regulation_identified=bool(app_raw.get("regulation_identified", False)),
            regulation_id=app_raw.get("regulation_id"),
            regulation_version=app_raw.get("regulation_version"),
            effective_date_iso=app_raw.get("effective_date_iso"),
            applicability=ApplicabilityStatus(
                app_raw.get("applicability", "UNKNOWN")
            ),
        )
    except (ValueError, KeyError, TypeError):
        return None


def _assess_field(
    key: str,
    raw: Any,
    req: FieldRequirement,
    has_content: bool,
) -> FieldAssessment:
    declared, conflicts = _extract_declared_state(raw)
    if not has_content:
        declared = EpistemicState.MISSING

    evidence_refs = _extract_evidence(raw)
    evidence_level = (
        EvidenceVerificationLevel.HASH_FORMAT_VALID
        if evidence_refs else EvidenceVerificationLevel.ABSENT
    )

    effective = declared
    downgrade_reason = ""

    if declared == EpistemicState.KNOWN:
        if req.requires_evidence:
            if not evidence_refs:
                effective = EpistemicState.OBSERVED
                downgrade_reason = "KNOWN_WITHOUT_EVIDENCE"
        else:
            effective = EpistemicState.OBSERVED
            downgrade_reason = "KNOWN_DECLARED_NOT_VERIFIED"

    applicability = None
    if req.requires_applicability:
        applicability = _extract_applicability(raw)
        if applicability is None:
            applicability = NormativeApplicability(
                jurisdiction_known=False, jurisdiction_id=None,
                source_identified=False, source_id=None,
                regulation_identified=False, regulation_id=None,
                regulation_version=None, effective_date_iso=None,
                applicability=ApplicabilityStatus.UNKNOWN,
            )
        if applicability.applicability == ApplicabilityStatus.CONFLICTING:
            effective = EpistemicState.CONFLICTING
            conflicts = conflicts + ("applicability_conflicting",)
        elif applicability.applicability == ApplicabilityStatus.UNKNOWN and req.blocking:
            effective = EpistemicState.UNKNOWN
            downgrade_reason = downgrade_reason or "APPLICABILITY_UNKNOWN"
        elif applicability.applicability == ApplicabilityStatus.NOT_APPLICABLE and req.blocking:
            effective = EpistemicState.MISSING
            downgrade_reason = downgrade_reason or "APPLICABILITY_NOT_APPLICABLE"

    return FieldAssessment(
        key=key,
        blocking=req.blocking,
        declared_state=declared,
        effective_state=effective,
        evidence_level=evidence_level,
        evidence_refs=evidence_refs,
        applicability=applicability,
        downgrade_reason=downgrade_reason,
        conflicts=conflicts,
    )


def _compute_proceed_allowed(
    status: SufficiencyStatus,
    profile: SufficiencyProfile,
    conditional_reasons: tuple[str, ...],
) -> bool:
    if status == SufficiencyStatus.SUFFICIENT:
        return True
    if status != SufficiencyStatus.CONDITIONALLY_SUFFICIENT:
        return False
    if not profile.allow_conditional:
        return False
    if "ASSUMED_BLOCKING" in conditional_reasons and not profile.allow_assumptions:
        return False
    return True


def _generate_questions(
    assessments: tuple[FieldAssessment, ...],
) -> tuple[HumanQuestion, ...]:
    questions: list[HumanQuestion] = []
    for a in assessments:
        if a.effective_state == EpistemicState.CONFLICTING and a.blocking:
            questions.append(HumanQuestion(
                key=a.key,
                question=f"Field '{a.key}' has conflicting values. Please arbitrate.",
                blocks_operation=True, priority=1,
                reason="CONFLICTING_BLOCKING",
                kind=QuestionKind.CONFLICT_ARBITRATION,
            ))
        elif a.effective_state == EpistemicState.MISSING and a.blocking:
            questions.append(HumanQuestion(
                key=a.key,
                question=f"Field '{a.key}' is missing and required.",
                blocks_operation=True, priority=1,
                reason="MISSING_BLOCKING",
                kind=QuestionKind.MISSING_FIELD,
            ))
        elif a.effective_state == EpistemicState.UNKNOWN and a.blocking:
            questions.append(HumanQuestion(
                key=a.key,
                question=f"Field '{a.key}' is UNKNOWN but required.",
                blocks_operation=True, priority=2,
                reason="UNKNOWN_BLOCKING",
                kind=QuestionKind.MISSING_FIELD,
            ))
        elif a.conflicts and not a.blocking:
            questions.append(HumanQuestion(
                key=a.key,
                question=f"Field '{a.key}' has conflicts (non-blocking). Please review.",
                blocks_operation=False, priority=3,
                reason="CONFLICTING_NON_BLOCKING",
                kind=QuestionKind.CONFLICT_ARBITRATION,
            ))
        if a.effective_state == EpistemicState.OBSERVED \
                and a.blocking \
                and a.downgrade_reason == "KNOWN_WITHOUT_EVIDENCE":
            questions.append(HumanQuestion(
                key=a.key,
                question=f"Field '{a.key}' declares KNOWN but lacks evidence.",
                blocks_operation=True, priority=2,
                reason="EVIDENCE_REQUIRED",
                kind=QuestionKind.EVIDENCE,
            ))
    return tuple(sorted(questions, key=lambda q: (q.priority, q.key)))


def check_input_sufficiency(
    snapshot: object,
    *,
    project_id: str,
    profile: SufficiencyProfile,
) -> SufficiencyReport:
    if not isinstance(project_id, str) or not project_id.strip():
        raise ValueError("project_id must be non-empty")
    if not isinstance(profile, SufficiencyProfile):
        raise TypeError("profile must be SufficiencyProfile")

    if not isinstance(snapshot, Mapping):
        return SufficiencyReport(
            contract_version=1, evaluation_id="",
            project_id=project_id, snapshot_fingerprint="", snapshot_version=0,
            profile_id=profile.profile_id, profile_version=profile.profile_version,
            operation=profile.operation,
            status=SufficiencyStatus.UNKNOWN,
            assessments=(), human_questions=(),
            reason="SNAPSHOT_NOT_INTERPRETABLE",
            proceed_allowed=False, assumptions_allowed=False,
            conditional_reasons=(), policy=profile.policy,
        )

    snapshot_fingerprint = _fingerprint_snapshot(snapshot)
    evaluation_id = _compute_evaluation_id(project_id, snapshot_fingerprint, profile)
    snapshot_version = int(snapshot.get("version", 0)) if isinstance(snapshot.get("version"), int) else 0

    assessments: list[FieldAssessment] = []
    for req in profile.requirements:
        raw = snapshot.get(req.key)
        assessments.append(_assess_field(req.key, raw, req, _has_content(raw)))

    blocking_assessments = [a for a in assessments if a.blocking]
    has_blocking_conflict = any(
        a.effective_state == EpistemicState.CONFLICTING for a in blocking_assessments
    )

    has_blocking_not_accepted = False
    has_assumed_blocking = False
    for a in blocking_assessments:
        req = next(r for r in profile.requirements if r.key == a.key)
        if a.effective_state in req.accepted_effective_states:
            if a.effective_state == EpistemicState.ASSUMED:
                has_assumed_blocking = True
        else:
            has_blocking_not_accepted = True

    non_blocking_issues = any(
        (a.effective_state in (EpistemicState.CONFLICTING, EpistemicState.MISSING))
        for a in assessments if not a.blocking
    )

    conditional_reasons: list[str] = []
    if has_blocking_conflict:
        status = SufficiencyStatus.CONFLICTING
    elif has_blocking_not_accepted:
        status = SufficiencyStatus.INSUFFICIENT
    elif has_assumed_blocking:
        status = SufficiencyStatus.CONDITIONALLY_SUFFICIENT
        conditional_reasons.append("ASSUMED_BLOCKING")
    elif non_blocking_issues:
        status = SufficiencyStatus.CONDITIONALLY_SUFFICIENT
        conditional_reasons.append("NON_BLOCKING_ISSUE")
    else:
        status = SufficiencyStatus.SUFFICIENT

    proceed_allowed = _compute_proceed_allowed(status, profile, tuple(conditional_reasons))
    if profile.require_human_authority:
        proceed_allowed = False

    assumptions_allowed = profile.allow_assumptions
    questions = _generate_questions(tuple(assessments))

    reason_map = {
        SufficiencyStatus.SUFFICIENT: "ALL_BLOCKING_SATISFIED",
        SufficiencyStatus.CONDITIONALLY_SUFFICIENT: (
            "CONDITIONAL_" + "_".join(conditional_reasons)
            if conditional_reasons else "CONDITIONAL"
        ),
        SufficiencyStatus.INSUFFICIENT: "BLOCKING_MISSING_OR_UNSATISFIED",
        SufficiencyStatus.CONFLICTING: "BLOCKING_CONFLICT_PRESERVED",
        SufficiencyStatus.UNKNOWN: "UNKNOWN",
    }

    return SufficiencyReport(
        contract_version=1,
        evaluation_id=evaluation_id,
        project_id=project_id,
        snapshot_fingerprint=snapshot_fingerprint,
        snapshot_version=snapshot_version,
        profile_id=profile.profile_id,
        profile_version=profile.profile_version,
        operation=profile.operation,
        status=status,
        assessments=tuple(assessments),
        human_questions=questions,
        reason=reason_map[status],
        proceed_allowed=proceed_allowed,
        assumptions_allowed=assumptions_allowed,
        conditional_reasons=tuple(conditional_reasons),
        policy=profile.policy,
        human_authority_ref=None,
        read_only=True,
    )

def _hex64(c="a"): return c * 64


def _policy():
    return PolicyIdentity(policy_id="p", policy_version=1, policy_fingerprint=_hex64())


def _req(key, evidence=False):
    return FieldRequirement(
        key=key, blocking=True,
        accepted_declared_states=frozenset({EpistemicState.KNOWN}),
        accepted_effective_states=frozenset({EpistemicState.KNOWN}),
        requires_evidence=evidence,
    )


def _profile(reqs, allow_cond=False, allow_ass=False):
    return SufficiencyProfile(
        profile_id="prof", profile_version=1,
        operation=Operation.RECOMPUTE_SUBGRAPH_V1,
        requirements=tuple(reqs),
        allow_conditional=allow_cond, allow_assumptions=allow_ass,
        policy=_policy(),
    )


def _ev():
    return {"id": "E", "kind": "doc", "content_hash": _hex64("b"),
            "source_id": "S", "declared_supports_field": "brief"}
