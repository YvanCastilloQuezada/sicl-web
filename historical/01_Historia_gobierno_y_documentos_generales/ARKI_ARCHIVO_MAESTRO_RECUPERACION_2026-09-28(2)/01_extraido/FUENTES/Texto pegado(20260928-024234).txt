3.3 Estructura del elemento
python
PropertyValue = str | int | float | bool

@dataclass(frozen=True)
class ArchiElement:
    element_id: ArchiElementId
    kind: ElementKind
    project_id: str
    name: str
    geometry: "ArchiGeometry"
    hosting: tuple[ArchiElementId, ...] = ()
    contained_in: ArchiElementId | None = None
    properties: tuple[tuple[str, PropertyValue], ...] = ()
    version: int = 1
    provenance: tuple[str, ...] = ()

    def __post_init__(self):
        if not self.name.strip(): raise ValueError("name required")
        if self.version < 1: raise ValueError("version >= 1")
        if self.hosting and self.contained_in is not None:
            raise ValueError("cannot host and be contained simultaneously")
        keys = [k for k, _ in self.properties]
        if keys != sorted(keys): raise ValueError("properties must be sorted")
        if len(keys) != len(set(keys)): raise ValueError("properties keys unique")

    def content_hash(self) -> str:
        state = {
            "schema": "archi-element-state-v1",
            "elementId": self.element_id.value,
            "kind": self.kind.value,
            "projectId": self.project_id,
            "name": self.name,
            "geometry": self.geometry.to_canonical(),
            "hosting": [h.value for h in self.hosting],
            "containedIn": self.contained_in.value if self.contained_in else None,
            "properties": [[k, _serialize_property_value(v)] for k, v in self.properties],
            "version": self.version,
        }
        canonical = json.dumps(state, sort_keys=True, separators=(",", ":"), ensure_ascii=False)
        return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
provenance fuera del content_hash. Cambio administrativo no altera identidad semántica.

3.4 Properties

def _serialize_property_value(v: PropertyValue) -> object:
    if isinstance(v, bool):   return {"t": "bool",  "v": v}
    if isinstance(v, int):    return {"t": "int",   "v": v}
    if isinstance(v, float):  return {"t": "float", "v": round(v, 6)}
    if isinstance(v, str):    return {"t": "str",   "v": v}
    raise TypeError(f"unsupported: {type(v)}")

El contrato H-002 congelado hace fail-closed para VERSION_UPGRADED e IDENTITY_CHANGED con razón NO_AUTOMATIC_RETROACTIVE_IMPACT. Su semántica es: "una versión superior no impacta retroactivamente a consumidores de versiones previas". Eso está bien.

El WHAT-IF pre-commit pregunta otra cosa: "si el contenido del elemento actual W-01@v cambia, ¿qué depende de W-01@v?" Eso es un cambio de contenido sobre la misma (entity_id, version), modalidad CONTENT_CHANGED. H-002 sí hace reverse traversal en ese caso.
