3. Modelo semántico
3.1 Identidad estable
python
@dataclass(frozen=True)
class ArchiElementId:
    """Stable architectural identity. Independent of mutable geometry."""
    value: str          # sha256 hex, 64 chars

    def __post_init__(self):
        if len(self.value) != 64 or any(c not in "0123456789abcdef" for c in self.value):
            raise ValueError("ArchiElementId must be 64 hex characters")

    @classmethod
    def compute(cls, project_id: str, kind: "ElementKind", birth_nonce: str) -> "ArchiElementId":
        canonical = json.dumps(
            {"schema": "archi-element-id-v1",
             "projectId": project_id,
             "kind": kind.value,
             "birthNonce": birth_nonce},
            sort_keys=True, separators=(",", ":"),
        )
        return cls(hashlib.sha256(canonical.encode("utf-8")).hexdigest())
Reglas duras:

birth_nonce es obligatorio. No se genera automáticamente.

Prohibido en el cálculo del ID: time.time(), Date.now(), uuid.uuid4(), os.urandom, hash de timestamp, hash de geometría.

El birth_nonce es responsabilidad del caller (opaco, reproducible, ej. "W-01").

Test obligatorio: dos invocaciones con mismo (project_id, kind, birth_nonce) → mismo ID.

Test obligatorio: mutar geometría de un ArchiElement no cambia su ArchiElementId.

@dataclass(frozen=True)
class IfcGlobalId:
    value: str          # 22 chars IFC base64

    def __post_init__(self):
        if len(self.value) != 22:
            raise ValueError("IfcGlobalId must be 22 characters")

    @classmethod
    def from_archi_id(cls, archi_id: ArchiElementId) -> "IfcGlobalId":
        """Deterministic: sha256(ArchiElementId) → base64 → 22 chars."""
        import base64
        digest = hashlib.sha256(archi_id.value.encode("utf-8")).digest()
        encoded = base64.b64encode(digest).decode("ascii")
        # IFC base64 variant: replace + / with _ $ for IFC compatibility
        ifc_b64 = encoded.translate(str.maketrans("+/", "_$"))
        return cls(ifc_b64[:22])

@dataclass(frozen=True)
class ArchiElement:
    element_id: ArchiElementId
    kind: ElementKind
    project_id: str
    name: str
    geometry: "ArchiGeometry"
    hosting: tuple[ArchiElementId, ...] = ()
    contained_in: ArchiElementId | None = None
    properties: tuple[tuple[str, PropertyValue], ...] = ()
    version: int = 1
    provenance: tuple[str, ...] = ()

    def __post_init__(self):
        if not self.name.strip():
            raise ValueError("name must be non-empty")
        if self.version < 1:
            raise ValueError("version must be positive")
        if self.hosting and self.contained_in is not None:
            raise ValueError("element cannot both host and be contained")
        # properties must be sorted by key
        keys = [k for k, _ in self.properties]
        if keys != sorted(keys):
            raise ValueError("properties must be sorted by key")
        if len(keys) != len(set(keys)):
            raise ValueError("properties keys must be unique")

class GeometryKind(str, Enum):
    EXTRUDED_RECTANGLE = "EXTRUDED_RECTANGLE"
    EXTRUDED_CIRCLE = "EXTRUDED_CIRCLE"

@dataclass(frozen=True)
class ProfileSpec:
    kind: GeometryKind
    width_mm: float | None = None       # rectangular
    height_mm: float | None = None      # rectangular
    radius_mm: float | None = None      # circular

    def __post_init__(self):
        if self.kind is GeometryKind.EXTRUDED_RECTANGLE:
            if self.width_mm is None or self.height_mm is None:
                raise ValueError("rectangle requires width and height")
            if self.width_mm <= 0 or self.height_mm <= 0:
                raise ValueError("dims must be positive")
        elif self.kind is GeometryKind.EXTRUDED_CIRCLE:
            if self.radius_mm is None or self.radius_mm <= 0:
                raise ValueError("circle requires positive radius")

@dataclass(frozen=True)
class ArchiGeometry:
    kind: GeometryKind
    origin_mm: tuple[float, float, float]
    direction: tuple[float, float, float]
    profile: ProfileSpec
    length_mm: float
    material_ref: str | None = None

    def __post_init__(self):
        if self.length_mm <= 0:
            raise ValueError("length must be positive")
        if len(self.origin_mm) != 3 or len(self.direction) != 3:
            raise ValueError("origin and direction must be 3-tuples")
        norm = sum(c * c for c in self.direction) ** 0.5
        if abs(norm - 1.0) > 1e-6:
            raise ValueError("direction must be unit vector")
Unidades: milímetros. Consistente con ARKI-DRAW y A-002.

class MutationKind(str, Enum):
    MOVE = "MOVE"
    RESIZE = "RESIZE"
    ADD_OPENING = "ADD_OPENING"
    DELETE = "DELETE"

@dataclass(frozen=True)
class ArchiMutation:
    kind: MutationKind
    target: ArchiElementId
    payload: tuple[tuple[str, PropertyValue], ...]  # ordenado por clave
    requested_by: str
    requested_at_iso: str                            # YYYY-MM-DDTHH:MM:SSZ
    mutation_id: str = ""                            # calculado si vacío

    def __post_init__(self):
        if not self.requested_by.strip():
            raise ValueError("requested_by must be non-empty")
        # ISO date check
        if len(self.requested_at_iso) != 20 or self.requested_at_iso[4] != "-" or self.requested_at_iso[10] != "T":
            raise ValueError("requested_at_iso must be YYYY-MM-DDTHH:MM:SSZ")

class MutationStatus(str, Enum):

@dataclass(frozen=True)
class MutationResult:
    status: MutationStatus
    mutation: ArchiMutation
    affected_elements: tuple[ArchiElementId, ...] = ()
    sufficiency: "SufficiencyReport | None" = None
    reaction_plan: "ReactionPlan | None" = None
    reason: str = ""
    provenance: tuple[str, ...] = ()

    def to_dict(self):
        return {
            "status": self.status.value,
            "mutationId": self.mutation.mutation_id,
            "affectedElements": [e.value for e in self.affected_elements],
            "reason": self.reason,
            "provenance": list(self.provenance),
        }
