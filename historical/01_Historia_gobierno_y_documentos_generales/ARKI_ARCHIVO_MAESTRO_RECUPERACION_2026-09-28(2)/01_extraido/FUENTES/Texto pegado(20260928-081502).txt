## 0. IDENTIDAD DEL PROYECTO

ARKI es un **arquitecto artificial multiescala**, no un chatbot, no un CAD, no un BIM, no un generador de imágenes. Es una **inteligencia arquitectónica computacional** que trata cada elemento (mesa, cuarto, casa, edificio, ciudad) como **ente vivo**: si algo cambia, todo cambia —área, metrados, representación, costo.

**Regla de oro:**
ONE ARCHITECTURAL STATE → MULTIPLE REPRESENTATIONS

text
Una sola verdad arquitectónica. Múltiples representaciones (2D, 3D, IFC, render) coherentes entre sí.

PARTE 2 — COMANDO PARA MANUS

bash
#!/usr/bin/env bash
set -euo pipefail

# ═══════════════════════════════════════════════════════════════
# ARKI — EMPAQUETADO DE SUPERPODERES
# Genera un ZIP consolidado con todas las capacidades activas
# ═══════════════════════════════════════════════════════════════

REPO=/home/ubuntu/sicl-core
OUT_DIR=/tmp/arki-superpowers
ZIP_NAME=arki-superpowers-2026-09-28.zip

# Verificar repo
cd "$REPO"
if [ ! -d .git ]; then
    echo "ERROR: $REPO no es un repositorio git"
    exit 1
fi

# Verificar baseline
CURRENT_HEAD=$(git rev-parse HEAD)
echo "HEAD actual: $CURRENT_HEAD"

# Limpiar directorios previos
rm -rf "$OUT_DIR"
mkdir -p "$OUT_DIR"

# ───────────────────────────────────────────────────────────────
# 1. Copiar CORE congelado desde main
# ───────────────────────────────────────────────────────────────
echo "→ Copiando CORE congelado desde main..."
git checkout main
git pull origin main 2>/dev/null || true
mkdir -p "$OUT_DIR/core"
cp -r src/sicl "$OUT_DIR/core/"
cp -r tests/test_derivation.py "$OUT_DIR/core/tests/" 2>/dev/null || true
cp -r tests/test_impact.py "$OUT_DIR/core/tests/" 2>/dev/null || true
cp -r tests/test_validity.py "$OUT_DIR/core/tests/" 2>/dev/null || true
cp -r tests/test_reaction.py "$OUT_DIR/core/tests/" 2>/dev/null || true

# Documentación de gobernanza
[ -f docs/ARKI_CORE_FROZEN_v1.0.md ] && cp docs/ARKI_CORE_FROZEN_v1.0.md "$OUT_DIR/core/"
[ -f docs/ARKI_KNOWN_LIMITATIONS_v1.0.md ] && cp docs/ARKI_KNOWN_LIMITATIONS_v1.0.md "$OUT_DIR/core/"

# ───────────────────────────────────────────────────────────────
# 2. Copiar ARKI-DRAW Bloque 1 desde su rama
# ───────────────────────────────────────────────────────────────
echo "→ Copiando ARKI-DRAW Bloque 1..."
mkdir -p "$OUT_DIR/arkidraw"
git show feat/arkidraw-block1:src/sicl/drawing/__init__.py > "$OUT_DIR/arkidraw/__init__.py" 2>/dev/null || echo "  (sin drawing/__init__.py)"
for f in primitives.py scale.py paper.py title_block.py sheet.py svg_export.py pdf_export.py ifc_synthetic.py; do
    git show feat/arkidraw-block1:src/sicl/drawing/$f > "$OUT_DIR/arkidraw/$f" 2>/dev/null || true
done
mkdir -p "$OUT_DIR/arkidraw/tests"
git show feat/arkidraw-block1:tests/test_drawing.py > "$OUT_DIR/arkidraw/tests/test_drawing.py" 2>/dev/null || true

# ───────────────────────────────────────────────────────────────
# 3. Copiar ARKI-3D D-2.1 desde su rama
# ───────────────────────────────────────────────────────────────
echo "→ Copiando ARKI-3D D-2.1..."
mkdir -p "$OUT_DIR/arki3d"
for f in __init__.py identity.py model.py relationships.py mutation.py sufficiency.py transaction.py ifc_export.py; do
    git show feat/arki3d-archi-model:src/sicl/archi/$f > "$OUT_DIR/arki3d/$f" 2>/dev/null || true
done
mkdir -p "$OUT_DIR/arki3d/tests"
for f in test_archi_identity.py test_archi_model.py test_archi_relationships.py test_archi_mutation.py test_archi_sufficiency.py test_archi_transaction.py test_archi_ifc_export.py test_archi_poc_d2.py; do
    git show feat/arki3d-archi-model:tests/$f > "$OUT_DIR/arki3d/tests/$f" 2>/dev/null || true
done

# ───────────────────────────────────────────────────────────────
# 4. Guardar el MANIFEST.md
# ───────────────────────────────────────────────────────────────
echo "→ Guardando MANIFEST.md..."
# IMPORTANTE: Manus debe pegar el contenido del MANIFEST entregado en este chat.
# Aquí se asume que ya existe en el repo o se guarda desde aquí.
cat > "$OUT_DIR/MANIFEST.md" << 'MANIFEST_EOF'
# ARKI — MANIFEST DE SUPERPODERES
# Generado: 2026-09-28
# (Contenido completo debe venir del MANIFEST entregado por DeepSeek)
MANIFEST_EOF

# ───────────────────────────────────────────────────────────────
# 5. Capturar identidad del paquete
# ───────────────────────────────────────────────────────────────
echo "→ Capturando identidad del paquete..."
{
    echo "# ARKI SUPERPOWERS PACKAGE — IDENTITY"
    echo "Fecha: $(date -u +%Y-%m-%dT%H:%M:%SZ)"
    echo ""
    echo "## Baselines"
    echo "core_main:       $(git rev-parse main)"
    echo "a002_branch:     $(git rev-parse feat/a002-v1.4-unified 2>/dev/null || echo 'N/A')"
    echo "arkidraw_branch: $(git rev-parse feat/arkidraw-block1 2>/dev/null || echo 'N/A')"
    echo "arki3d_branch:   $(git rev-parse feat/arki3d-archi-model 2>/dev/null || echo 'N/A')"
    echo ""
    echo "## Tags vigentes"
    git tag -l | sort
} > "$OUT_DIR/IDENTITY.md"

# ───────────────────────────────────────────────────────────────
# 6. Hashes SHA-256 por archivo
# ───────────────────────────────────────────────────────────────
echo "→ Calculando hashes por archivo..."
find "$OUT_DIR" -type f -exec sha256sum {} \; | sed "s|$OUT_DIR/||" > "$OUT_DIR/SHA256SUMS.txt"

# ───────────────────────────────────────────────────────────────
# 7. Volver a main (estado limpio)
# ───────────────────────────────────────────────────────────────
git checkout main

# ───────────────────────────────────────────────────────────────
# 8. Crear ZIP
# ───────────────────────────────────────────────────────────────
echo "→ Creando ZIP..."
cd /tmp
zip -r "$ZIP_NAME" arki-superpowers/

# ───────────────────────────────────────────────────────────────
# 9. Hash del ZIP
# ───────────────────────────────────────────────────────────────
echo ""
echo "═══════════════════════════════════════════════════════"
echo "PAQUETE CREADO"
echo "═══════════════════════════════════════════════════════"
echo "Ruta:      /tmp/$ZIP_NAME"
echo "Tamaño:    $(du -h /tmp/$ZIP_NAME | cut -f1)"
echo "SHA-256:   $(sha256sum /tmp/$ZIP_NAME | cut -d' ' -f1)"
echo "═══════════════════════════════════════════════════════"
