@dataclass(frozen=True)
class ArchiElementId:
    """Stable identity. Independent of mutable geometry."""
    value: str          # sha256 hex, 64 chars

    @classmethod
    def compute(cls, project_id: str, kind: "ElementKind", birth_nonce: str) -> "ArchiElementId":
        canonical = json.dumps(
            {"schema": "archi-element-id-v1",
             "projectId": project_id,
             "kind": kind.value,
             "birthNonce": birth_nonce},
            sort_keys=True, separators=(",", ":"),
        )
        return cls(hashlib.sha256(canonical.encode("utf-8")).hexdigest())

Mismo (project_id, kind, birth_nonce) → mismo ID.

Mover geometría NO cambia el ID.

@dataclass(frozen=True)
class IfcGlobalId:
    value: str          # 22 chars IFC base64

    @classmethod
    def from_archi_id(cls, archi_id: ArchiElementId) -> "IfcGlobalId":
        import base64
        digest = hashlib.sha256(archi_id.value.encode("utf-8")).digest()
        encoded = base64.b64encode(digest).decode("ascii").translate(str.maketrans("+/", "_$"))
        return cls(encoded[:22])

3.3 Estructura del elemento
python
PropertyValue = str | int | float | bool

@dataclass(frozen=True)
class ArchiElement:
    element_id: ArchiElementId
    kind: ElementKind
    project_id: str
    name: str
    geometry: "ArchiGeometry"
    hosting: tuple[ArchiElementId, ...] = ()
    contained_in: ArchiElementId | None = None
    properties: tuple[tuple[str, PropertyValue], ...] = ()
    version: int = 1
    provenance: tuple[str, ...] = ()

    def __post_init__(self):
        if not self.name.strip(): raise ValueError("name required")
        if self.version < 1: raise ValueError("version must be >= 1")
        if self.hosting and self.contained_in is not None:
            raise ValueError("cannot host and be contained simultaneously")
        keys = [k for k, _ in self.properties]
        if keys != sorted(keys): raise ValueError("properties must be sorted")
        if len(keys) != len(set(keys)): raise ValueError("properties keys unique")

    def content_hash(self) -> str:
        """Deterministic SHA-256 of the canonical semantic state."""
        state = {
            "schema": "archi-element-state-v1",
            "elementId": self.element_id.value,
            "kind": self.kind.value,
            "projectId": self.project_id,
            "name": self.name,
            "geometry": self.geometry.to_canonical(),
            "hosting": [h.value for h in self.hosting],
            "containedIn": self.contained_in.value if self.contained_in else None,
            "properties": [[k, _serialize_property_value(v)] for k, v in self.properties],
            "version": self.version,
        }
        canonical = json.dumps(state, sort_keys=True, separators=(",", ":"), ensure_ascii=False)
        return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
Regla: provenance queda FUERA del content_hash. Cambiar provenance administrativa no altera identidad semántica.

3.4 Properties tagged union
python
def _serialize_property_value(v: PropertyValue) -> object:
    if isinstance(v, bool):   return {"t": "bool",  "v": v}
    if isinstance(v, int):    return {"t": "int",   "v": v}
    if isinstance(v, float):  return {"t": "float", "v": round(v, 6)}
    if isinstance(v, str):    return {"t": "str",   "v": v}
    raise TypeError(f"unsupported property value: {type(v)}")
bool comprobado antes que int (porque isinstance(True, int) es True). Floats canonicalizados a 6 decimales.

5.3 Identidad y versionado de relaciones
Cambio de geometría de source/target → NO cambia la identidad ni la versión de la relación. La relación semántica persiste.

Cambio de source, target o RelationshipKind → cambio estructural: nuevo relationship_id (la identidad depende de source+target+kind). La relación anterior se cierra.

Añadir/eliminar provenance → no cambia identidad ni versión.

6.1 Contrato de mutación

class MutationKind(str, Enum):
    MOVE = "MOVE"
    RESIZE = "RESIZE"
    ADD_OPENING = "ADD_OPENING"
    DELETE = "DELETE"

@dataclass(frozen=True)
class ArchiMutation:
    kind: MutationKind
    target: ArchiElementId
    target_version: int
    payload: tuple[tuple[str, PropertyValue], ...]     # ordenado por clave
    requested_by: str
    requested_at_iso: str                              # provenance, NO identity
    request_nonce: str                                 # caller-provided, required
    project_id: str
    mutation_id: str = ""                              # computed if empty

    def __post_init__(self):
        if not self.requested_by.strip(): raise ValueError("requested_by required")
        if not self.request_nonce.strip(): raise ValueError("request_nonce required")
        if not self.project_id.strip(): raise ValueError("project_id required")
        keys = [k for k, _ in self.payload]
        if keys != sorted(keys): raise ValueError("payload must be sorted")

    def compute_id(self) -> str:
        identity = {
            "schema": "archi-mutation-id-v1",
            "projectId": self.project_id,
            "mutationKind": self.kind.value,
            "targetId": self.target.value,
            "targetVersion": self.target_version,
            "canonicalPayload": [[k, _serialize_property_value(v)] for k, v in self.payload],
            "requestNonce": self.request_nonce,
        }
        canonical = json.dumps(identity, sort_keys=True, separators=(",", ":"))
        return hashlib.sha256(canonical.encode("utf-8")).hexdigest()

    def with_id(self) -> "ArchiMutation":
        if self.mutation_id:
            return self
        return replace(self, mutation_id=self.compute_id())

6.5 Versionado unificado
ArchiElement.version es la verdad canónica.

VersionedRef.version debe ser exactamente igual a ArchiElement.version.

Mutación semántica aplicada: version(n+1) = version(n) + 1.

Cambio únicamente de provenance: no incrementa versión.

No hay dos contadores independientes.

6.6 Hosting semantics

Un hosted nunca queda flotando en silencio.
