4. Una primera conversación con ARKI
Antes de integrar voz y visión, podemos probar el cerebro local mediante un programa pequeño. Este ejemplo utiliza un modelo que ya tengas descargado y no modifica ARKI.


import json
import urllib.request

MODELO = "REEMPLAZAR_CON_MODELO_INSTALADO"

def preguntar_arki(pregunta):
    datos = {
        "model": MODELO,
        "stream": False,
        "messages": [
            {
                "role": "system",
                "content": (
                    "Eres un asistente experimental de ARKI. "
                    "Ayudas a estudiar arquitectura y programación. "
                    "No afirmes haber ejecutado herramientas "
                    "ni modificado proyectos."
                )
            },
            {"role": "user", "content": pregunta}
        ]
    }

    solicitud = urllib.request.Request(
        "http://127.0.0.1:11434/api/chat",
        data=json.dumps(datos).encode("utf-8"),
        headers={"Content-Type": "application/json"}
    )

    with urllib.request.urlopen(
        solicitud, timeout=300
    ) as respuesta:
        resultado = json.load(respuesta)

    return resultado["message"]["content"]

Guárdalo como prueba_arki_ollama.py, reemplaza REEMPLAZAR_CON_MODELO_INSTALADO por un nombre de ollama list y ejecútalo con:

python3 prueba_arki_ollama.py

Esto demuestra una conversación local básica. Todavía no es el ARKI integrado ni tiene memoria persistente, voz o visión.

import json, time, urllib.request, subprocess

BASE = "http://127.0.0.1:11434"
MODELS = [
    "llama3.2-vision:11b",
    "qwen3-coder:30b",
    "devstral-small-2:24b",
    "qwen3-coder-next:q4_K_M",
]

def api(path, payload=None, timeout=180):
    data = None if payload is None else json.dumps(payload).encode()
    req = urllib.request.Request(
        BASE + path, data=data,
        headers={"Content-Type": "application/json"}
    )
    with urllib.request.urlopen(req, timeout=timeout) as response:
        return json.load(response)

for model in MODELS:
    print("\n" + "=" * 70, flush=True)
    print("MODELO:", model, flush=True)
    start = time.monotonic()
    try:
        result = api("/api/chat", {
            "model": model,
            "messages": [{"role": "user", "content":
                "Responde exactamente: ARKI LOCAL OK"}],
            "stream": False,
            "options": {"num_ctx": 2048, "num_predict": 40},
            "keep_alive": 0
        }, timeout=240)
        print("RESPUESTA:", result.get("message", {}).get("content", "")[:400])
        print("SEGUNDOS:", round(time.monotonic() - start, 2))
        print("TOKENS GENERADOS:", result.get("eval_count", "UNKNOWN"))
        print("ESTADO: PASS" if result.get("done") else "ESTADO: INCOMPLETE")
    except Exception as exc:
        print("ESTADO: FAIL")
        print("ERROR:", type(exc).__name__, str(exc)[:500])
    finally:
        try:
            api("/api/generate", {"model": model, "keep_alive": 0})
        except Exception:
            pass
        subprocess.run(["ollama", "stop", model], capture_output=True)

print("\n=== GPU FINAL ===", flush=True)
subprocess.run(["nvidia-smi", "--query-gpu=name,memory.used,memory.total",
                "--format=csv,noheader"])
print("\n=== OLLAMA FINAL ===", flush=True)
